    ```python
    def test_not_found_renders_info_and_hides_evidence(monkeypatch):
        from src.llm.generate import QueryAnswer

        def fake_run(*args, **kwargs):
            return QueryAnswer(
                prose="",
                evidence=[],
                not_found=True,
                suggestion="Try asking about T9 compatibility or wiring.",
            )

        # Patch the symbol imported into app.py's namespace
        import app
        monkeypatch.setattr(app, "run_query_structured", fake_run)

        at = AppTest.from_file(APP_PATH).run(timeout=10)
        at.text_input("Your question").set_value("what is the color").run(timeout=10)
        at.button("Ask").click().run(timeout=10)

        # NOTE: monkeypatch applied to the imported module must survive AppTest rerun.
        # AppTest re-executes app.py on each .run(), so patch must be re-applied.
        # See pytest-monkeypatch docs — module-level monkeypatch survives within a
        # single test function. If AppTest reimports app freshly, fall back to
        # monkeypatching run_query_structured where it is defined (src.pipeline.query)
        # instead of on the app module.

        assert any("Try asking about T9 compatibility" in i.value for i in at.info)
        assert len(at.expander) == 0  # D-11: evidence expander hidden in not-found state
    ```

    ```python
        at = AppTest.from_file(APP_PATH).run(timeout=10)
        at.text_input("Your question").set_value("T9 accessories?").run(timeout=10)

        expander_labels = [e.label for e in at.expander]
    ```
