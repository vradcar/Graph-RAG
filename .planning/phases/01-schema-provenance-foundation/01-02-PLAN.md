```python
# src/graph/provenance.py — required public API:
from typing import Any, Dict
from neo4j import ManagedTransaction

def merge_document(tx: ManagedTransaction, doc: Dict[str, Any]) -> None:
    """Upsert a :Document node. doc must contain keys: doc_id, title, sku, source_url, ingested_at."""

def merge_node_with_provenance(
    tx: ManagedTransaction,
    label: str,           # validated against VALID_KINDS by caller
    node_id: str,
    props: Dict[str, Any],
    doc_id: str,          # REQUIRED — never optional
) -> None:
    """MERGE node by id; accumulate source_docs via coll.distinct; MERGE MENTIONED_IN to :Document {doc_id}."""

def merge_edge_with_provenance(
    tx: ManagedTransaction,
    source_id: str,
    target_id: str,
    rel: str,             # validated against VALID_RELATIONS by caller
    props: Dict[str, Any],
    doc_id: str,          # REQUIRED — keyed in MERGE map
) -> None:
    """MERGE (a)-[r:REL {source_doc: $doc_id}]->(b); ON CREATE/ON MATCH SET r += props."""
```

|----------|-------------|

| graph_items.json → loader | Trusted local file; produced by ingest.py upstream. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-04 | Tampering | Relationship type from edge JSON | mitigate | Existing safe_rel sanitization (alnum + underscore + uppercase) is preserved; additionally validate against VALID_RELATIONS where possible and log unknown types. |
| T-01-05 | Tampering | --doc-id from CLI flowing into Cypher | mitigate | doc_id is parameterized ($doc_id) in every helper — never f-string substituted. |
| T-01-06 | Information Disclosure | source_doc value reveals filename slug | accept | Slug is the intended citation surface (SCHEMA-03 / QUALITY-03). Not sensitive. |
| T-01-07 | Repudiation | Re-ingest overwriting first_ingested | mitigate | merge_document uses ON CREATE for first_ingested (set once) and ON MATCH for last_reingested (overwrite). Test test_document_upsert asserts this. |
| T-01-08 | Denial of Service | source_docs[] growing unbounded on repeat re-ingest | mitigate | coll.distinct guard in merge_node_with_provenance; test test_idempotent_reingest asserts source_docs length stays at 1 across N re-ingests of the same doc. |
</threat_model>
